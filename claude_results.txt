import java.util.Collections;
import java.util.HashSet;
import java.util.Set;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class CosmicCats {

    private final Set<String> catIdentifiers;

    public CosmicCats() {
        Set<String> identifiers = new HashSet<>();
        identifiers.add("Whiskers");
        identifiers.add("Paws");
        identifiers.add("Claw");
        this.catIdentifiers = Collections.unmodifiableSet(identifiers);
    }

    public boolean isCat(String identifier) {
        return catIdentifiers.contains(identifier);
    }

    public String listCats() {
        return catIdentifiers.toString();
    }

    public static void main(String[] args) throws InterruptedException {
        final CosmicCats cosmicCats = new CosmicCats();

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable checkerTask = new Runnable() {
            @Override
            public void run() {
                String[] candidates = {"Whiskers", "Paws", "Claw", "Fluffy", "Tiger"};
                for (String candidate : candidates) {
                    System.out.println(Thread.currentThread().getName() + " checking " + candidate + ": "
                            + cosmicCats.isCat(candidate));
                }
            }
        };

        Runnable listerTask = new Runnable() {
            @Override
            public void run() {
                System.out.println(Thread.currentThread().getName() + " cat list: "
                        + cosmicCats.listCats());
            }
        };

        executor.submit(checkerTask);
        executor.submit(listerTask);

        executor.shutdown();
        executor.awaitTermination(5, TimeUnit.SECONDS);

        System.out.println("Final cat list: " + cosmicCats.listCats());
    }
}

import java.util.Collections;
import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class DelegatingDeviceTracker {

    private final ConcurrentHashMap<String, Position> gadgetPositions;
    private final Map<String, Position> unmodifiableLiveView;

    public DelegatingDeviceTracker(Map<String, Position> initialPositions) {
        this.gadgetPositions = new ConcurrentHashMap<>(initialPositions);
        this.unmodifiableLiveView = Collections.unmodifiableMap(gadgetPositions);
    }

    public Position getPosition(String gadgetId) {
        return gadgetPositions.get(gadgetId);
    }

    public boolean updatePosition(String gadgetId, int x, int y) {
        Position newPosition = new Position(x, y);
        Position previous = gadgetPositions.computeIfPresent(gadgetId, (key, oldValue) -> newPosition);
        return previous != null;
    }

    public Map<String, Position> getLivePositionsView() {
        return unmodifiableLiveView;
    }

    public Map<String, Position> getPositionsSnapshot() {
        return Collections.unmodifiableMap(new HashMap<>(gadgetPositions));
    }

    static class Position {
        final int x;
        final int y;

        Position(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public String toString() {
            return "Position{x=" + x + ", y=" + y + '}';
        }
    }

    public static void main(String[] args) throws InterruptedException {
        Map<String, Position> initialMap = new HashMap<>();
        initialMap.put("gadget-1", new Position(0, 0));
        initialMap.put("gadget-2", new Position(5, 5));
        initialMap.put("gadget-3", new Position(10, 10));

        final DelegatingDeviceTracker tracker = new DelegatingDeviceTracker(initialMap);

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable updaterTask = new Runnable() {
            @Override
            public void run() {
                for (int i = 0; i < 5; i++) {
                    tracker.updatePosition("gadget-1", i, i * 2);
                    tracker.updatePosition("gadget-2", i + 10, i + 20);
                    try {
                        Thread.sleep(10);
                    } catch (InterruptedException e) {
                        Thread.currentThread().interrupt();
                    }
                }
            }
        };

        Runnable readerTask = new Runnable() {
            @Override
            public void run() {
                for (int i = 0; i < 5; i++) {
                    Position position = tracker.getPosition("gadget-1");
                    System.out.println("Reader sees gadget-1: " + position);
                    System.out.println("Live view: " + tracker.getLivePositionsView());
                    try {
                        Thread.sleep(10);
                    } catch (InterruptedException e) {
                        Thread.currentThread().interrupt();
                    }
                }
            }
        };

        executor.submit(updaterTask);
        executor.submit(readerTask);

        executor.shutdown();
        executor.awaitTermination(5, TimeUnit.SECONDS);

        boolean modified = tracker.updatePosition("gadget-nonexistent", 0, 0);
        System.out.println("Modification of nonexistent gadget succeeded: " + modified);

        System.out.println("Final snapshot: " + tracker.getPositionsSnapshot());
        System.out.println("Final live view: " + tracker.getLivePositionsView());
    }
}

import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.ThreadLocalRandom;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.ReentrantLock;

public class LockerMover {

    static class ParcelCount {
        private int items;

        ParcelCount(int initialItems) {
            this.items = initialItems;
        }

        int getItems() {
            return items;
        }

        void add(int delta) {
            items += delta;
        }
    }

    static class Locker {
        private final String id;
        private final ReentrantLock lock = new ReentrantLock();
        private final ParcelCount parcelCount;

        Locker(String id, int initialItems) {
            this.id = id;
            this.parcelCount = new ParcelCount(initialItems);
        }

        String getId() {
            return id;
        }

        ReentrantLock getLock() {
            return lock;
        }

        ParcelCount getParcelCount() {
            return parcelCount;
        }
    }

    static class NotEnoughItemsException extends Exception {
        NotEnoughItemsException(String message) {
            super(message);
        }
    }

    private static final long FIXED_DELAY_MILLIS = 15L;
    private static final long RANDOM_DELAY_BOUND_MILLIS = 25L;

    private long calculateFixedDelay() {
        return FIXED_DELAY_MILLIS;
    }

    private long calculateRandomDelay() {
        return ThreadLocalRandom.current().nextLong(0, RANDOM_DELAY_BOUND_MILLIS);
    }

    public boolean move(Locker source, Locker target, int quantity, long timeout, TimeUnit unit)
            throws NotEnoughItemsException, InterruptedException {

        long deadlineNanos = System.nanoTime() + unit.toNanos(timeout);

        while (System.nanoTime() < deadlineNanos) {
            boolean sourceLocked = false;
            boolean targetLocked = false;

            try {
                long remainingNanos = deadlineNanos - System.nanoTime();
                if (remainingNanos <= 0) {
                    return false;
                }

                sourceLocked = source.getLock().tryLock(
                        Math.min(remainingNanos, TimeUnit.MILLISECONDS.toNanos(calculateFixedDelay())),
                        TimeUnit.NANOSECONDS);

                if (sourceLocked) {
                    remainingNanos = deadlineNanos - System.nanoTime();
                    if (remainingNanos <= 0) {
                        return false;
                    }

                    targetLocked = target.getLock().tryLock(
                            Math.min(remainingNanos, TimeUnit.MILLISECONDS.toNanos(calculateFixedDelay())),
                            TimeUnit.NANOSECONDS);

                    if (targetLocked) {
                        if (source.getParcelCount().getItems() < quantity) {
                            throw new NotEnoughItemsException(
                                    "Locker " + source.getId() + " does not have enough items to move " + quantity);
                        }
                        source.getParcelCount().add(-quantity);
                        target.getParcelCount().add(quantity);
                        return true;
                    }
                }
            } finally {
                if (targetLocked) {
                    target.getLock().unlock();
                }
                if (sourceLocked) {
                    source.getLock().unlock();
                }
            }

            long backoffMillis = calculateFixedDelay() + calculateRandomDelay();
            long remainingNanos = deadlineNanos - System.nanoTime();
            if (remainingNanos <= 0) {
                return false;
            }
            Thread.sleep(Math.min(backoffMillis, TimeUnit.NANOSECONDS.toMillis(remainingNanos) + 1));
        }

        return false;
    }

    public static void main(String[] args) throws InterruptedException {
        final Locker lockerA = new Locker("LockerA", 50);
        final Locker lockerB = new Locker("LockerB", 50);
        final LockerMover mover = new LockerMover();

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable taskAtoB = new Runnable() {
            @Override
            public void run() {
                try {
                    boolean result = mover.move(lockerA, lockerB, 10, 2, TimeUnit.SECONDS);
                    System.out.println("Move A->B result: " + result
                            + " | LockerA items: " + lockerA.getParcelCount().getItems()
                            + ", LockerB items: " + lockerB.getParcelCount().getItems());
                } catch (NotEnoughItemsException e) {
                    System.out.println("Move A->B failed: " + e.getMessage());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    System.out.println("Move A->B interrupted.");
                }
            }
        };

        Runnable taskBtoA = new Runnable() {
            @Override
            public void run() {
                try {
                    boolean result = mover.move(lockerB, lockerA, 5, 2, TimeUnit.SECONDS);
                    System.out.println("Move B->A result: " + result
                            + " | LockerA items: " + lockerA.getParcelCount().getItems()
                            + ", LockerB items: " + lockerB.getParcelCount().getItems());
                } catch (NotEnoughItemsException e) {
                    System.out.println("Move B->A failed: " + e.getMessage());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    System.out.println("Move B->A interrupted.");
                }
            }
        };

        executor.execute(taskAtoB);
        executor.execute(taskBtoA);

        executor.shutdown();
        executor.awaitTermination(10, TimeUnit.SECONDS);

        System.out.println("Final LockerA items: " + lockerA.getParcelCount().getItems());
        System.out.println("Final LockerB items: " + lockerB.getParcelCount().getItems());
    }
}

    static class Container {
        private final String id;
        private final ReentrantLock lock = new ReentrantLock();
        private final ElementCount elementCount;

        Container(String id, int initialElements) {
            this.id = id;
            this.elementCount = new ElementCount(initialElements);
        }

        String getId() {
            return id;
        }

        ReentrantLock getLock() {
            return lock;
        }

        ElementCount getElementCount() {
            return elementCount;
        }
    }

    static class InsufficientElementsException extends Exception {
        InsufficientElementsException(String message) {
            super(message);
        }
    }

    private static final long FIXED_DELAY_MILLIS = 15L;
    private static final long RANDOM_DELAY_BOUND_MILLIS = 25L;

    private long calculateFixedDelay() {
        return FIXED_DELAY_MILLIS;
    }

    private long calculateRandomDelay() {
        return ThreadLocalRandom.current().nextLong(0, RANDOM_DELAY_BOUND_MILLIS);
    }

    public boolean swap(Container origin, Container target, int numberOfElements, long timeout, TimeUnit unit)
            throws InsufficientElementsException, InterruptedException {

        long deadlineNanos = System.nanoTime() + unit.toNanos(timeout);

        while (System.nanoTime() < deadlineNanos) {
            boolean originLocked = false;
            boolean targetLocked = false;

            try {
                long remainingNanos = deadlineNanos - System.nanoTime();
                if (remainingNanos <= 0) {
                    return false;
                }

                originLocked = origin.getLock().tryLock(
                        Math.min(remainingNanos, TimeUnit.MILLISECONDS.toNanos(calculateFixedDelay())),
                        TimeUnit.NANOSECONDS);

                if (originLocked) {
                    remainingNanos = deadlineNanos - System.nanoTime();
                    if (remainingNanos <= 0) {
                        return false;
                    }

                    targetLocked = target.getLock().tryLock(
                            Math.min(remainingNanos, TimeUnit.MILLISECONDS.toNanos(calculateFixedDelay())),
                            TimeUnit.NANOSECONDS);

                    if (targetLocked) {
                        if (origin.getElementCount().getElements() < numberOfElements) {
                            throw new InsufficientElementsException(
                                    "Container " + origin.getId() + " does not have enough elements to swap "
                                            + numberOfElements);
                        }
                        origin.getElementCount().add(-numberOfElements);
                        target.getElementCount().add(numberOfElements);
                        return true;
                    }
                }
            } finally {
                if (targetLocked) {
                    target.getLock().unlock();
                }
                if (originLocked) {
                    origin.getLock().unlock();
                }
            }

            long backoffMillis = calculateFixedDelay() + calculateRandomDelay();
            long remainingNanos = deadlineNanos - System.nanoTime();
            if (remainingNanos <= 0) {
                return false;
            }
            Thread.sleep(Math.min(backoffMillis, TimeUnit.NANOSECONDS.toMillis(remainingNanos) + 1));
        }

        return false;
    }

    public static void main(String[] args) throws InterruptedException {
        final Container containerA = new Container("ContainerA", 50);
        final Container containerB = new Container("ContainerB", 50);
        final ContainerSwapper swapper = new ContainerSwapper();

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable taskAtoB = new Runnable() {
            @Override
            public void run() {
                try {
                    boolean result = swapper.swap(containerA, containerB, 10, 2, TimeUnit.SECONDS);
                    System.out.println("Swap A->B result: " + result
                            + " | ContainerA elements: " + containerA.getElementCount().getElements()
                            + ", ContainerB elements: " + containerB.getElementCount().getElements());
                } catch (InsufficientElementsException e) {
                    System.out.println("Swap A->B failed: " + e.getMessage());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    System.out.println("Swap A->B interrupted.");
                }
            }
        };

        Runnable taskBtoA = new Runnable() {
            @Override
            public void run() {
                try {

                    System.out.println("Swap B->A result: " + result

                } catch (InterruptedException e) {

                    System.out.println("Swap B->A interrupted.");
                }
            }
        };

        }

        String getId() {
            return id;

        ReentrantLock getLock() {

        }

            return widgetCount;
        }

    static class NotEnoughWidgetsException extends Exception {
        NotEnoughWidgetsException(String message) {
            super(message);
        }
    }

    private static final long FIXED_DELAY_MILLIS = 15L;
    private static final long RANDOM_DELAY_BOUND_MILLIS = 25L;

    private long calculateFixedDelay() {
        return FIXED_DELAY_MILLIS;
    }

    private long calculateRandomDelay() {
        return ThreadLocalRandom.current().nextLong(0, RANDOM_DELAY_BOUND_MILLIS);
    }

    public boolean move(Bin source, Bin target, int numberOfWidgets, long timeout, TimeUnit unit)
            throws NotEnoughWidgetsException, InterruptedException {

        long deadlineNanos = System.nanoTime() + unit.toNanos(timeout);

        while (System.nanoTime() < deadlineNanos) {

                            Math.min(remainingNanos, TimeUnit.MILLISECONDS.toNanos(calculateFixedDelay())),

                        if (source.getWidgetCount().getWidgets() < numberOfWidgets) {
                            throw new NotEnoughWidgetsException(
                                    "Bin " + source.getId() + " does not have enough widgets to move "

    }

    public int channelTheGrimoire(List<Rune> candidateRunes, Grimoire grimoire) {
        List<Rune> transmutedRunes = new ArrayList<Rune>(candidateRunes);
        Collections.sort(transmutedRunes, new RuneComparator());

        boolean[] used = new boolean[transmutedRunes.size()];
        int harmonyCount = 0;

        for (int i = 0; i < transmutedRunes.size(); i++) {
            if (used[i]) {
                continue;
            }
            Rune current = transmutedRunes.get(i);
            for (int j = i + 1; j < transmutedRunes.size(); j++) {
                if (used[j]) {
                    continue;

                    used[i] = true;
                    used[j] = true;
                    Harmony harmony = new Harmony(current, candidate);
                    grimoire.crystallize(harmony);
                    harmonyCount++;
                    break;
                }
            }
        }

        return harmonyCount;
    }

final class ConstellationHarmonyTask implements Callable<List<Harmony>> {
    private final List<Rune> assignedRunes;
    private final Comparator<Rune> comparator;

    ConstellationHarmonyTask(List<Rune> assignedRunes, Comparator<Rune> comparator) {
        this.assignedRunes = assignedRunes;
        this.comparator = comparator;
    }

    @Override
    public List<Harmony> call() {
        List<Rune> localCopy = new ArrayList<Rune>(assignedRunes);
        Collections.sort(localCopy, comparator);

        Map<Constellation, List<Rune>> localGroups = new EnumMap<Constellation, List<Rune>>(Constellation.class);
        for (Rune rune : localCopy) {
            List<Rune> group = localGroups.get(rune.getConstellation());
            if (group == null) {
                group = new ArrayList<Rune>();
                localGroups.put(rune.getConstellation(), group);
            }
            group.add(rune);
        }

        List<Harmony> localHarmonies = new ArrayList<Harmony>();
        for (List<Rune> group : localGroups.values()) {
            List<Rune> positives = new ArrayList<Rune>();
            List<Rune> negatives = new ArrayList<Rune>();
            for (Rune rune : group) {
                if (rune.getResonance() == Resonance.POSITIVE) {
                    positives.add(rune);
                } else {
                    negatives.add(rune);
                }
            }
            int pairCount = Math.min(positives.size(), negatives.size());
            for (int i = 0; i < pairCount; i++) {
                localHarmonies.add(new Harmony(positives.get(i), negatives.get(i)));
            }
        }
        return localHarmonies;
    }
}

        Map<Constellation, List<Rune>> groupedByConstellation = new EnumMap<Constellation, List<Rune>>(Constellation.class);
        for (Rune rune : confinedCandidates) {
            List<Rune> group = groupedByConstellation.get(rune.getConstellation());
            if (group == null) {
                group = new ArrayList<Rune>();
                groupedByConstellation.put(rune.getConstellation(), group);

            group.add(rune);

        int splitIndex = (constellationKeys.size() + 1) / 2;

        List<Rune> firstChunk = new ArrayList<Rune>();
        List<Rune> secondChunk = new ArrayList<Rune>();

        for (int i = 0; i < constellationKeys.size(); i++) {
            List<Rune> group = groupedByConstellation.get(constellationKeys.get(i));
            if (i < splitIndex) {
                firstChunk.addAll(group);
            } else {
                secondChunk.addAll(group);
            }
        }

        ExecutorService executorService = Executors.newFixedThreadPool(2);
        List<Harmony> allHarmonies = new ArrayList<Harmony>();

                    new ConstellationHarmonyTask(secondChunk, transmutationComparator));

            allHarmonies.addAll(firstFuture.get());
            allHarmonies.addAll(secondFuture.get());
        } finally {
            executorService.shutdown();
            executorService.awaitTermination(5, TimeUnit.SECONDS);
        }

        grimoire.crystallize(allHarmonies);
        return grimoire.totalHarmonies();
    }

final class Grimoire {

    void crystallize(List<Harmony> harmonies) {
        crystallizedHarmonies.addAll(harmonies);
    }

    int totalHarmonies() {
        return crystallizedHarmonies.size();
    }

        synchronized (crystallizedHarmonies) {

final class RuneComparator implements Comparator<Rune> {
    @Override
    public int compare(Rune a, Rune b) {
        int constellationCompare = a.getConstellation().compareTo(b.getConstellation());
        if (constellationCompare != 0) {
            return constellationCompare;
        }
        return a.getResonance().compareTo(b.getResonance());
    }
}

final class ConstellationHarmonyTask implements Callable<List<Harmony>> {
    private final List<Rune> assignedRunes;
    private final Comparator<Rune> comparator;

    ConstellationHarmonyTask(List<Rune> assignedRunes, Comparator<Rune> comparator) {
        this.assignedRunes = assignedRunes;
        this.comparator = comparator;
    }

    @Override
    public List<Harmony> call() {
        List<Rune> localCopy = new ArrayList<Rune>(assignedRunes);
        Collections.sort(localCopy, comparator);

        Map<Constellation, List<Rune>> localGroups = new EnumMap<Constellation, List<Rune>>(Constellation.class);
        for (Rune rune : localCopy) {
            List<Rune> group = localGroups.get(rune.getConstellation());
            if (group == null) {
                group = new ArrayList<Rune>();
                localGroups.put(rune.getConstellation(), group);
            }
            group.add(rune);
        }

        List<Harmony> localHarmonies = new ArrayList<Harmony>();
        for (List<Rune> group : localGroups.values()) {
            List<Rune> positives = new ArrayList<Rune>();
            List<Rune> negatives = new ArrayList<Rune>();
            for (Rune rune : group) {
                if (rune.getResonance() == Resonance.POSITIVE) {
                    positives.add(rune);
                } else {
                    negatives.add(rune);
                }
            }
            int pairCount = Math.min(positives.size(), negatives.size());
            for (int i = 0; i < pairCount; i++) {
                localHarmonies.add(new Harmony(positives.get(i), negatives.get(i)));
            }
        }
        return localHarmonies;
    }
}

public class RuneGrimoireHarmonizer {

    private final Grimoire grimoire = new Grimoire();

    int channelTheGrimoire(List<Rune> candidateRunes) throws Exception {
        final Comparator<Rune> transmutationComparator = new RuneComparator();

        List<Rune> confinedCandidates = new ArrayList<Rune>(candidateRunes);

        Map<Constellation, List<Rune>> groupedByConstellation = new EnumMap<Constellation, List<Rune>>(Constellation.class);
        for (Rune rune : confinedCandidates) {
            List<Rune> group = groupedByConstellation.get(rune.getConstellation());
            if (group == null) {
                group = new ArrayList<Rune>();
                groupedByConstellation.put(rune.getConstellation(), group);
            }
            group.add(rune);
        }

        List<Constellation> constellationKeys = new ArrayList<Constellation>(groupedByConstellation.keySet());
        int splitIndex = (constellationKeys.size() + 1) / 2;

        List<Rune> firstChunk = new ArrayList<Rune>();
        List<Rune> secondChunk = new ArrayList<Rune>();

        for (int i = 0; i < constellationKeys.size(); i++) {
            List<Rune> group = groupedByConstellation.get(constellationKeys.get(i));
            if (i < splitIndex) {
                firstChunk.addAll(group);
            } else {
                secondChunk.addAll(group);
            }
        }

        ExecutorService executorService = Executors.newFixedThreadPool(2);
        List<Harmony> allHarmonies = new ArrayList<Harmony>();

        try {
            Future<List<Harmony>> firstFuture = executorService.submit(
                    new ConstellationHarmonyTask(firstChunk, transmutationComparator));
            Future<List<Harmony>> secondFuture = executorService.submit(
                    new ConstellationHarmonyTask(secondChunk, transmutationComparator));

            allHarmonies.addAll(firstFuture.get());
            allHarmonies.addAll(secondFuture.get());
        } finally {
            executorService.shutdown();
            executorService.awaitTermination(5, TimeUnit.SECONDS);
        }

        grimoire.crystallize(allHarmonies);
        return grimoire.totalHarmonies();
    }

    Grimoire getGrimoire() {
        return grimoire;
    }

abstract class DreamWeaver {

    private final FutureTask<Vision> futureTask;
    private final Thread pilotThread;

    DreamWeaver() {
        Callable<Vision> callable = new Callable<Vision>() {
            @Override
            public Vision call() throws VisionCastException {
                return loadVision();
            }
        };
        this.futureTask = new FutureTask<Vision>(callable);
        this.pilotThread = new Thread(futureTask);
    }

    public abstract Vision loadVision() throws VisionCastException;

    public void start() {
        pilotThread.start();
    }

    public Vision get() throws VisionCastException {
        try {
            return futureTask.get();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new VisionCastException("Vision summoning was interrupted", e);
        } catch (ExecutionException e) {
            Throwable cause = e.getCause();
            if (cause instanceof VisionCastException) {
                throw (VisionCastException) cause;
            }
            throw new VisionCastException("Unexpected error during vision summoning", cause);
        }
    }
}

import java.util.concurrent.ExecutorService;

import java.util.concurrent.locks.ReentrantLock;

public class VaultTransferSystem {

    interface GoldAmount {
        long getValue();
        GoldAmount add(GoldAmount other);
        GoldAmount subtract(GoldAmount other);
        boolean isLessThan(GoldAmount other);
    }

    static class GoldAmountImpl implements GoldAmount {
        private final long value;

        GoldAmountImpl(long value) {
            if (value < 0) {
                throw new IllegalArgumentException("Gold amount cannot be negative");
            }
            this.value = value;
        }

        @Override
        public long getValue() {
            return value;
        }

        @Override
        public GoldAmount add(GoldAmount other) {
            return new GoldAmountImpl(this.value + other.getValue());
        }

        @Override
        public GoldAmount subtract(GoldAmount other) {
            return new GoldAmountImpl(this.value - other.getValue());
        }

        @Override
        public boolean isLessThan(GoldAmount other) {
            return this.value < other.getValue();
        }

        @Override
        public String toString() {
            return String.valueOf(value);
        }
    }

    interface Vault {
        String getName();
        GoldAmount getBalance();
        void debit(GoldAmount amount) throws InsufficientTreasureException;
        void credit(GoldAmount amount);
        Lock getLock();
    }

    static class VaultImpl implements Vault {
        private final String name;
        private GoldAmount balance;
        private final Lock lock = new ReentrantLock();

        VaultImpl(String name, GoldAmount initialBalance) {
            this.name = name;
            this.balance = initialBalance;
        }

        @Override
        public String getName() {
            return name;
        }

        @Override
        public GoldAmount getBalance() {
            return balance;
        }

        @Override
        public void debit(GoldAmount amount) throws InsufficientTreasureException {
            if (balance.isLessThan(amount)) {
                throw new InsufficientTreasureException(
                        "Vault " + name + " has insufficient treasure. Required: "
                                + amount + ", Available: " + balance);
            }
            balance = balance.subtract(amount);
        }

        @Override
        public void credit(GoldAmount amount) {
            balance = balance.add(amount);
        }

        @Override
        public Lock getLock() {
            return lock;
        }
    }

    static class InsufficientTreasureException extends Exception {
        InsufficientTreasureException(String message) {
            super(message);
        }
    }

    static class TransferRitual {
        private static final Object TIE_BREAK_LOCK = new Object();

        void transfer(Vault from, Vault to, GoldAmount amount) throws InsufficientTreasureException {
            int fromHash = System.identityHashCode(from);
            int toHash = System.identityHashCode(to);

            if (fromHash == toHash) {
                synchronized (TIE_BREAK_LOCK) {
                    lockAndTransfer(from, to, amount);
                }
            } else {
                lockAndTransfer(from, to, amount);
            }
        }

        private void lockAndTransfer(Vault from, Vault to, GoldAmount amount) throws InsufficientTreasureException {
            int fromHash = System.identityHashCode(from);
            int toHash = System.identityHashCode(to);

            Vault first = fromHash <= toHash ? from : to;
            Vault second = fromHash <= toHash ? to : from;

            first.getLock().lock();
            try {
                second.getLock().lock();
                try {
                    from.debit(amount);
                    to.credit(amount);
                    System.out.println(Thread.currentThread().getName()
                            + " transferred " + amount + " gold from "
                            + from.getName() + " to " + to.getName());
                } finally {
                    second.getLock().unlock();
                }
            } finally {
                first.getLock().unlock();
            }
        }
    }

    public static void main(String[] args) throws InterruptedException {
        Vault vaultA = new VaultImpl("VaultA", new GoldAmountImpl(1000));
        Vault vaultB = new VaultImpl("VaultB", new GoldAmountImpl(1000));

        TransferRitual ritual = new TransferRitual();

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable transferAtoB = () -> {
            for (int i = 0; i < 5; i++) {
                try {
                    ritual.transfer(vaultA, vaultB, new GoldAmountImpl(50));
                } catch (InsufficientTreasureException e) {
                    System.out.println("Transfer A->B failed: " + e.getMessage());
                }
            }
        };

        Runnable transferBtoA = () -> {
            for (int i = 0; i < 5; i++) {
                try {
                    ritual.transfer(vaultB, vaultA, new GoldAmountImpl(30));
                } catch (InsufficientTreasureException e) {
                    System.out.println("Transfer B->A failed: " + e.getMessage());
                }
            }
        };

        executor.submit(transferAtoB);
        executor.submit(transferBtoA);

        executor.shutdown();
        executor.awaitTermination(30, TimeUnit.SECONDS);

        System.out.println("Final balance of VaultA: " + vaultA.getBalance());
        System.out.println("Final balance of VaultB: " + vaultB.getBalance());
    }

        private final Orb[] buffer;
        private final int capacity;
        private int putIndex = 0;
        private int takeIndex = 0;

        private final Semaphore availableOrbs;
        private final Semaphore availableSlots;
        private final Object lock = new Object();

        CircularVault(int capacity) {
            this.capacity = capacity;
            this.buffer = new Orb[capacity];
            this.availableOrbs = new Semaphore(0);
            this.availableSlots = new Semaphore(capacity);
        }

        void put(Orb x) throws InterruptedException {
            availableSlots.acquire();
            synchronized (lock) {
                buffer[putIndex] = x;
                putIndex = (putIndex + 1) % capacity;
                System.out.println(Thread.currentThread().getName() + " placed " + x + " into the vault");

            availableOrbs.acquire();
            Orb result;

                System.out.println(Thread.currentThread().getName() + " retrieved " + result + " from the vault");
            }
            availableSlots.release();
            return result;
        }

        boolean isEmpty() {
            synchronized (lock) {
                return availableOrbs.availablePermits() == 0;
            }
        }

        boolean isFull() {
            synchronized (lock) {
                return availableSlots.availablePermits() == 0;
            }
        }

        boolean containsValue(V value) {

public class SheepCounter {

    private volatile boolean asleep = false;
    private int sheepCount = 0;

    public void tryToSleep() {
        while (!asleep) {
            countSomeSheep();
        }
        System.out.println("Finally asleep after counting " + sheepCount + " sheep.");
    }

    private void countSomeSheep() {
        sheepCount++;
        try {
            Thread.sleep(50);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    public void wakeUpAndSleep() {
        asleep = true;
    }

    public static void main(String[] args) throws InterruptedException {

        ExecutorService executor = Executors.newFixedThreadPool(2);

        Runnable sleepTask = new Runnable() {

            public void run() {
                sheepCounter.tryToSleep();
            }
        };

            @Override
            public void run() {
                try {
                    Thread.sleep(500);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();

                sheepCounter.wakeUpAndSleep();
            }
        };

        executor.submit(sleepTask);
        executor.submit(alarmTask);

        executor.shutdown();
